計測スクリプト（再現用）

import os, time
now = time.time()
cats = {}
for root, dirs, files in os.walk('.'):
    if '.git' in root or 'node_modules' in root: continue
    for f in files:
        if not f.endswith('.html'): continue
        path = os.path.join(root, f).replace(os.sep, '/').lstrip('./')
        cat = path.split('/')[0] if '/' in path else '(root)'
        with open(path, encoding='utf-8', errors='ignore') as fh:
            lines = sum(1 for _ in fh)
        days_old = (now - os.path.getmtime(path)) / 86400
        cats.setdefault(cat, []).append((path, lines, days_old))
for c, files in sorted(cats.items()):
    total = sum(f[1] for f in files)
    ages = [f[2] for f in files]
    print(f"{c:<20}{len(files):>6}p {total:>7}lines  avg={sum(ages)/len(ages):.1f}d")

これを logs/hub-stats.py として置き、四半期ごとに実行 → このページを手動更新する想定。将来的には GitHub Actions で自動化できる。
